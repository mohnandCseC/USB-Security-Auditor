@'
[CmdletBinding()]
param()

$ScriptDir = [System.AppDomain]::CurrentDomain.BaseDirectory
if ([string]::IsNullOrWhiteSpace($ScriptDir)) { $ScriptDir = (Get-Location).Path }

Clear-Host
Write-Host "=== USB ADVANCED SECURITY AUDITOR & TRIAGE ===" -ForegroundColor Green
Write-Host "[*] Gathering system components and digital artifacts..." -ForegroundColor Cyan

$Timestamp = Get-Date -Format "yyyy-MM-dd_HH-mm-ss"
$ReportBase = "Audit_$($env:COMPUTERNAME)_$Timestamp"
$JsonPath = [System.IO.Path]::Combine($ScriptDir, "$ReportBase.json")
$HtmlPath = [System.IO.Path]::Combine($ScriptDir, "$ReportBase.html")

# 1. System & Hardware Details
$OS = (Get-CimInstance Win32_OperatingSystem).Caption
$CPU = (Get-CimInstance Win32_Processor | Select-Object -First 1).Name
$RAM_GB = [math]::Round((Get-CimInstance Win32_ComputerSystem).TotalPhysicalMemory / 1GB, 2)
$User = [System.Security.Principal.WindowsIdentity]::GetCurrent().Name
$BootTime = (Get-CimInstance Win32_OperatingSystem).LastBootUpTime
$Uptime = (Get-Date) - $BootTime
$UptimeStr = "$($Uptime.Days)d $($Uptime.Hours)h $($Uptime.Minutes)m"

$DiskC = Get-CimInstance Win32_LogicalDisk -Filter "DeviceID='C:'"
$DiskFreeGB = [math]::Round($DiskC.FreeSpace / 1GB, 2)
$DiskTotalGB = [math]::Round($DiskC.Size / 1GB, 2)

# 2. Network Information
$Gateway = (Get-NetRoute -DestinationPrefix '0.0.0.0/0' -ErrorAction SilentlyContinue | Sort-Object RouteMetric | Select-Object -First 1).NextHop
$ActiveIdx = (Get-NetRoute -DestinationPrefix '0.0.0.0/0' -ErrorAction SilentlyContinue | Sort-Object RouteMetric | Select-Object -First 1).InterfaceIndex
$PrimaryIP = "Disconnected"
$PrimaryMAC = "N/A"
if ($ActiveIdx) {
    $PrimaryIP = (Get-NetIPAddress -InterfaceIndex $ActiveIdx -AddressFamily IPv4 -ErrorAction SilentlyContinue).IPAddress
    $PrimaryMAC = (Get-NetAdapter -InterfaceIndex $ActiveIdx -ErrorAction SilentlyContinue).MacAddress
}
$DnsServers = ((Get-DnsClientServerAddress -InterfaceIndex $ActiveIdx -AddressFamily IPv4 -ErrorAction SilentlyContinue).ServerAddresses) -join ', '
$WifiSSID = try { ((netsh wlan show interfaces) -match '^\s*SSID\s*:\s*(.+)$' -replace '^\s*SSID\s*:\s*', '').Trim() } catch { "None" }

# 3. Security Baseline Engine
$Score = 100
$SecChecks = [ordered]@{}

# Defender
$Defender = try { Get-MpComputerStatus -ErrorAction Stop } catch { $null }
$DefActive = $Defender.RealTimeProtectionEnabled -and $Defender.AntivirusEnabled
$SecChecks["Defender Realtime Protection"] = if ($DefActive) { "ENABLED" } else { $Score -= 15; "DISABLED" }

$SigAge = if ($Defender.AntivirusSignatureAge) { $Defender.AntivirusSignatureAge } else { 999 }
if ($SigAge -le 7) {
    $SecChecks["Defender Signatures"] = "UP-TO-DATE ($SigAge days old)"
} else {
    $Score -= 5
    $SecChecks["Defender Signatures"] = "OUTDATED ($SigAge days old)"
}

# Firewall
$FwCount = (Get-NetFirewallProfile -ErrorAction SilentlyContinue | Where-Object { -not $_.Enabled }).Count
$SecChecks["Firewall (All Profiles)"] = if ($FwCount -eq 0) { "ENABLED" } else { $Score -= 10; "DISABLED" }

# BitLocker
$BitLocker = try { (Get-BitLockerVolume -MountPoint "C:" -ErrorAction Stop).ProtectionStatus } catch { "Off" }
$SecChecks["BitLocker (C:)"] = if ($BitLocker -eq "On") { "ENCRYPTED" } else { $Score -= 10; "UNENCRYPTED" }

# Secure Boot
$SecureBoot = try { Confirm-SecureBootUEFI } catch { $false }
$SecChecks["Secure Boot"] = if ($SecureBoot) { "ENABLED" } else { $Score -= 10; "DISABLED" }

# Guest Account
$GuestDisabled = try { (Get-CimInstance Win32_UserAccount -Filter "Name='Guest'").Disabled } catch { $true }
$SecChecks["Guest Account"] = if ($GuestDisabled) { "DISABLED" } else { $Score -= 5; "ENABLED" }

# SMBv1
$Smb = (Get-WindowsOptionalFeature -Online -FeatureName SMB1Protocol -ErrorAction SilentlyContinue).State
$SecChecks["SMBv1 Protocol"] = if ($Smb -eq "Enabled") { $Score -= 15; "ENABLED" } else { "DISABLED" }

# RDP
$RdpVal = (Get-ItemProperty "HKLM:\System\CurrentControlSet\Control\Terminal Server" -Name fDenyTSConnections -ErrorAction SilentlyContinue).fDenyTSConnections
$SecChecks["Remote Desktop (RDP)"] = if ($RdpVal -eq 0) { $Score -= 5; "ENABLED" } else { "DISABLED" }

# LLMNR Protocol
$LLMNR = (Get-ItemProperty "HKLM:\Software\Policies\Microsoft\Windows NT\DNSClient" -Name "EnableMulticast" -ErrorAction SilentlyContinue).EnableMulticast
$SecChecks["LLMNR Protocol"] = if ($LLMNR -eq 0) { "DISABLED" } else { $Score -= 5; "ENABLED (Poisoning Risk)" }

# Remote Registry
$RemoteReg = (Get-Service -Name "RemoteRegistry" -ErrorAction SilentlyContinue).Status
$SecChecks["Remote Registry Service"] = if ($RemoteReg -eq "Running") { $Score -= 5; "RUNNING" } else { "STOPPED" }

# PowerShell Execution Policy
$Policy = Get-ExecutionPolicy
$SecChecks["System Execution Policy"] = if ($Policy -in @("Unrestricted", "Bypass")) { $Score -= 5; "$Policy (Permissive)" } else { "$Policy" }

# Sensitive Ports
$OpenPorts = Get-NetTCPConnection -State Listen -ErrorAction SilentlyContinue | Select-Object -Unique LocalPort
$SensitivePorts = @(21, 22, 23, 445, 3389)
$FoundRiskyPorts = $OpenPorts | Where-Object { $SensitivePorts -contains $_.LocalPort } | Select-Object -ExpandProperty LocalPort
if ($FoundRiskyPorts) {
    $Score -= 10
    $SecChecks["Sensitive Listening Ports"] = "EXPOSED (" + ($FoundRiskyPorts -join ', ') + ")"
} else {
    $SecChecks["Sensitive Listening Ports"] = "CLEAN"
}

$FinalScore = [math]::Max(0, $Score)

# 4. Artifact Gathering
$StartupRaw = Get-CimInstance Win32_StartupCommand -ErrorAction SilentlyContinue | Where-Object { $_.Location -notmatch 'S-1-5-18|S-1-5-19|S-1-5-20' }
$CleanStartup = @()
foreach ($item in $StartupRaw) {
    $loc = if ($item.Location -match 'HKU\\S-1-5-21') { "Current User (HKCU)" } 
           elseif ($item.Location -match 'Common') { "All Users (HKLM)" }
           elseif ($item.Location -match 'Startup') { "Startup Folder" }
           else { "Registry AutoRun" }
    $CleanStartup += [PSCustomObject]@{ Name=$item.Name; Command=$item.Command; Location=$loc }
}
$CleanStartup = $CleanStartup | Sort-Object Name -Unique
$StartupRows = ($CleanStartup | ForEach-Object { "<tr><td><b>$($_.Name)</b></td><td style='font-family:Consolas;'>$($_.Command)</td><td><span class='badge' style='background:#334155;color:#e2e8f0;'>$($_.Location)</span></td></tr>" }) -join ""

$UsbHistory = Get-ItemProperty -Path "HKLM:\SYSTEM\CurrentControlSet\Enum\USBSTOR\*" -ErrorAction SilentlyContinue | Select-Object FriendlyName, PSChildName
$UsbHistRows = ($UsbHistory | ForEach-Object { "<tr><td>$($_.FriendlyName)</td><td>$($_.PSChildName)</td></tr>" }) -join ""

$DnsCache = Get-DnsClientCache -ErrorAction SilentlyContinue | Select-Object -First 20 -Unique Entry, Data
$DnsCacheRows = ($DnsCache | ForEach-Object { "<tr><td>$($_.Entry)</td><td>$($_.Data)</td></tr>" }) -join ""

$HostsPath = "$env:windir\System32\drivers\etc\hosts"
$HostsContent = Get-Content $HostsPath -ErrorAction SilentlyContinue | Where-Object { $_ -notmatch '^\s*#' -and $_.Trim() -ne '' }
$HostsRows = if ($HostsContent) {
    ($HostsContent | ForEach-Object { "<tr><td style='color:#f87171;font-weight:bold;'>$_</td></tr>" }) -join ""
} else {
    "<tr><td>Standard Windows default hosts file (Clean)</td></tr>"
}

$SuspiciousTasks = Get-ScheduledTask -ErrorAction SilentlyContinue | Where-Object { 
    $_.State -ne 'Disabled' -and 
    $_.Actions.Execute -and 
    $_.Actions.Execute -notmatch '^%windir%|^C:\\Windows' 
} | Select-Object TaskName, @{N='Path';E={$_.Actions.Execute}}, State | Select-Object -First 10
$TaskRows = ($SuspiciousTasks | ForEach-Object { "<tr><td><b>$($_.TaskName)</b></td><td style='font-family:Consolas;'>$($_.Path)</td><td>$($_.State)</td></tr>" }) -join ""

$Admins = try { (Get-LocalGroupMember -Group "Administrators" -ErrorAction SilentlyContinue).Name } catch { "Access Denied" }
$AdminRows = ($Admins | ForEach-Object { "<tr><td>$_</td></tr>" }) -join ""

$UsbList = Get-PnpDevice -PresentOnly -Class 'USB' -ErrorAction SilentlyContinue | Select-Object FriendlyName, InstanceId
$UsbRows = ($UsbList | ForEach-Object { "<tr><td>$($_.FriendlyName)</td><td>$($_.InstanceId)</td></tr>" }) -join ""

$ListeningConns = Get-NetTCPConnection -State Listen -ErrorAction SilentlyContinue | Select-Object LocalAddress, LocalPort, OwningProcess | Select-Object -First 15
$PortRows = ($ListeningConns | ForEach-Object { "<tr><td>$($_.LocalAddress):$($_.LocalPort)</td><td>$((Get-Process -Id $_.OwningProcess -ErrorAction SilentlyContinue).ProcessName) (PID: $($_.OwningProcess))</td></tr>" }) -join ""

$Hotfixes = Get-CimInstance Win32_QuickFixEngineering -ErrorAction SilentlyContinue | Sort-Object InstalledOn -Descending | Select-Object -First 5
$HotfixRows = ($Hotfixes | ForEach-Object { "<tr><td>$($_.HotFixID)</td><td>$($_.Description)</td><td>$($_.InstalledOn)</td></tr>" }) -join ""

# 5. CLI Output Summary
Write-Host "Computer       : $env:COMPUTERNAME"
Write-Host "Primary IP     : $PrimaryIP"
Write-Host "MAC Address    : $PrimaryMAC"
Write-Host "----------------------------------"
foreach ($chk in $SecChecks.Keys) {
    $val = $SecChecks[$chk]
    $color = if ($val -match "ENABLED|CLEAN|ENCRYPTED|UP-TO-DATE|STOPPED" -and $chk -notmatch "SMBv1|Remote Desktop|Guest|LLMNR") { "Green" }
             elseif ($chk -match "SMBv1|Remote Desktop|Guest|LLMNR" -and $val -match "DISABLED") { "Green" }
             else { "Red" }
    Write-Host "[*] $chk : $val" -ForegroundColor $color
}
Write-Host "----------------------------------"
Write-Host "Security Score : $FinalScore/100" -ForegroundColor Cyan

# 6. Export JSON
$FullData = [ordered]@{
    System      = @{ Computer=$env:COMPUTERNAME; User=$User; OS=$OS; CPU=$CPU; RAM_GB=$RAM_GB; Uptime=$UptimeStr; Storage="Free: $DiskFreeGB GB / Total: $DiskTotalGB GB" }
    Network     = @{ PrimaryIP=$PrimaryIP; MAC=$PrimaryMAC; Gateway=$Gateway; DNS=$DnsServers; WifiSSID=$WifiSSID }
    Audit       = @{ SecurityScore=$FinalScore; Checks=$SecChecks }
    Admins      = $Admins
    Startup     = $CleanStartup
    USB_Present = $UsbList
    USB_History = $UsbHistory
    DNS_Cache   = $DnsCache
    Hosts_File  = $HostsContent
    Tasks       = $SuspiciousTasks
}
$FullData | ConvertTo-Json -Depth 4 | Out-File -FilePath $JsonPath -Encoding utf8

$ScoreColor = if($FinalScore -ge 80){'#22c55e'}elseif($FinalScore -ge 50){'#eab308'}else{'#ef4444'}

$SecCheckRows = ""
foreach ($k in $SecChecks.Keys) {

    $badge = if ($isPass) { "<span class='badge pass' data-en='PASS' data-ar='آمن'>PASS</span>" } else { "<span class='badge fail' data-en='ALERT' data-ar='تنبيه'>ALERT</span>" }
    $SecCheckRows += "<tr><td>$k</td><td>$v</td><td>$badge</td></tr>"
}
